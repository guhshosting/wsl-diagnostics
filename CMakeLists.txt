cmake_minimum_required(VERSION 3.24)
project(wsl-diagnostics VERSION 1.0.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows and the MSVC toolchain are required")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(diagnostics src/core.cpp src/windows.cpp)
target_include_directories(diagnostics PUBLIC src)
target_compile_definitions(diagnostics PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(diagnostics PUBLIC advapi32 version shell32 ole32)
add_executable(wsl-diagnostics src/main.cpp src/version.rc src/app.manifest)
target_link_libraries(wsl-diagnostics PRIVATE diagnostics)
foreach(target diagnostics wsl-diagnostics)
  target_compile_options(${target} PRIVATE /W4 /WX /analyze /utf-8 /permissive- /sdl /EHsc)
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(wsl-diagnostics-tests tests/tests.cpp)
  target_link_libraries(wsl-diagnostics-tests PRIVATE diagnostics)
  target_compile_options(wsl-diagnostics-tests PRIVATE /W4 /WX /analyze /utf-8 /permissive- /sdl /EHsc)
  add_test(NAME unit COMMAND wsl-diagnostics-tests)
  add_test(NAME help COMMAND wsl-diagnostics --help)
  set_tests_properties(help PROPERTIES PASS_REGULAR_EXPRESSION "read-only")
  add_test(NAME version COMMAND wsl-diagnostics --version)
  set_tests_properties(version PROPERTIES PASS_REGULAR_EXPRESSION "1.0.0")
  add_test(NAME cli-contract COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli.ps1 -Exe $<TARGET_FILE:wsl-diagnostics>)
  set_tests_properties(unit help version cli-contract PROPERTIES TIMEOUT 30)
  add_executable(wsl-diagnostics-permissions tests/permissions.cpp)
  target_compile_options(wsl-diagnostics-permissions PRIVATE /W4 /WX /analyze /utf-8 /permissive- /sdl /EHsc)
  target_link_libraries(wsl-diagnostics-permissions PRIVATE diagnostics)
  add_test(NAME ordinary-user COMMAND wsl-diagnostics-permissions)
  set_tests_properties(ordinary-user PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 30)
endif()
install(TARGETS wsl-diagnostics RUNTIME DESTINATION .)
install(FILES README.md LICENSE DESTINATION .)
install(FILES docs/JSON.md docs/VALIDATION.md DESTINATION docs)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_FILE_NAME "wsl-diagnostics-1.0.0-windows-x64")
include(CPack)
